236:     public static boolean isCanonicalPath(String value) {
237:         if (value == null || value.isBlank()) {
238:             return false;
239:         }
240:         if (value.indexOf('\0') >= 0 || value.contains("\\")) {
241:             return false;
242:         }
243:         if (value.trim().length() != value.length()) {
244:             return false;
245:         }
246:         if (value.endsWith("/") || value.matches("^[A-Za-z]:/.*")) {
247:             return false;
248:         }
249:         try {
250:             var path = Path.of(value);
251:             if (path.isAbsolute()) {
252:                 return false;
253:             }
254:             for (var segment : path) {
255:                 var name = segment.toString();
256:                 if ("..".equals(name) || ".".equals(name)
257:                         || ".agent".equals(name) || ".git".equals(name)) {
258:                     return false;
259:                 }
260:             }
261:             var canonical = new StringJoiner("/");
262:             for (var segment : path.normalize()) {
263:                 canonical.add(segment.toString());
264:             }
265:             return path.getNameCount() > 0 && canonical.toString().equals(value);
