151:     private AcpxUsage readCurrentUsage(Path transcriptPath, AcpxPromptCompletion completion) throws IOException {
152:         if (!Files.isRegularFile(transcriptPath)) return null;
153:         AcpxStreamWireLine.Result result = null;
154:         BigDecimal sessionCost = null;
155:         String currency = "";
156:         int responses = 0;
157:         for (var line : Files.readAllLines(transcriptPath)) {
158:             if (line.isBlank()) continue;
159:             var wire = AcpxStreamWireLine.parse(line);
160:             if (line.stripLeading().startsWith("{") && wire.kind() == AcpxStreamWireLine.Kind.MALFORMED)
161:                 return AcpxUsage.unavailable("INVALID_CURRENT_TRANSCRIPT");
162:             if (isPrompt(wire) && !matches(wire, completion))
163:                 return AcpxUsage.unavailable("CURRENT_PROMPT_IDENTITY_MISMATCH");
164:             if (isUsageUpdate(wire) && completion.sessionId().equals(wire.notification().sessionId())) {
165:                 var update = wire.notification().usageUpdate();
166:                 if (update.costAmount() != null) {
167:                     sessionCost = update.costAmount();
168:                     currency = update.costCurrency();
169:                 }
170:             }
171:             if (wire.kind() == AcpxStreamWireLine.Kind.JSONRPC_ERROR
172:                     && completion.requestId().equals(wire.requestId()))
173:                 return AcpxUsage.unavailable("CURRENT_PROMPT_ERROR");
174:             if (wire.kind() != AcpxStreamWireLine.Kind.JSONRPC_OK || wire.result() == null) continue;
175:             var candidate = wire.result();
176:             if (candidate.stopReason().isBlank() && candidate.usage() == null
177:                     && !completion.requestId().equals(wire.requestId())) continue;
178:             if (!completion.requestId().equals(wire.requestId())
179:                     || !completion.stopReason().equals(candidate.stopReason())
180:                     || (!candidate.sessionId().isBlank() && !completion.sessionId().equals(candidate.sessionId())))
181:                 return AcpxUsage.unavailable("CURRENT_PROMPT_IDENTITY_MISMATCH");
182:             if (++responses > 1) return AcpxUsage.unavailable("AMBIGUOUS_CURRENT_PROMPT_RESULT");
183:             result = candidate;
