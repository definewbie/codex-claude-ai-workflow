package dev.dual.adapter.acpx;

import com.fasterxml.jackson.databind.ObjectMapper;
import dev.dual.core.id.DualId;
import dev.dual.core.workflow.*;
import dev.dual.ledger.file.*;
import java.nio.file.*;
import java.security.MessageDigest;
import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicReference;

public final class PhaseANetworkGrantLauncher {
    static final ObjectMapper JSON = new ObjectMapper();
    static final WorkflowFileLedger LEDGER = new WorkflowFileLedger();
    static final ExecutionNetworkGrantStore STORE = new ExecutionNetworkGrantStore();

    public static void main(String[] args) throws Exception {
        if (args.length != 3 || !List.of("prepare", "run").contains(args[0]))
            throw new IllegalArgumentException("usage: prepare|run <empty-workspace> <fixture-dir>");
        var root = Path.of(args[1]).toRealPath();
        check(root.startsWith(Path.of("/private/tmp")), "disposable /private/tmp workspace required");
        try (var entries = Files.list(root)) { check(entries.findAny().isEmpty(), "workspace must be empty"); }
        var fixture = Path.of(args[2]).toRealPath();
        var evidence = root.getParent();
        var paths = setup(root);
        var node = active(paths);
        var run = "run-" + UUID.randomUUID();
        var session = "phase-a-" + UUID.randomUUID();
        var runDir = Files.createDirectories(paths.runsDir().resolve(run));
        var grantRef = node.inputTransition().get(ExecutionNetworkGrantStore.REF_KEY).toString();
        var pendingSha = sha(root.resolve(grantRef));
        var prepared = ClaudeExecutionPolicy.write(root, paths.workstreamId().value(),
                node.nodeInstanceId(), run, runDir, AcpxPermissionProfile.implement(), null);
        assertPolicy(root, paths, prepared, List.of());
        check(pendingSha.equals(sha(root.resolve(grantRef))), "prepare consumed grant");
        check(JSON.readTree(root.resolve(grantRef).toFile()).path("state").asText().equals("PENDING"), "pending state");
        JSON.writerWithDefaultPrettyPrinter().writeValue(evidence.resolve("prepare.json").toFile(), Map.of(
                "workstream", paths.workstreamId().value(), "node", node.nodeInstanceId(), "run", run,
                "session", session, "grantRef", grantRef, "pendingGrantSha256", pendingSha,
                "policy", prepared.metadata("prepare_only")));
        Files.writeString(evidence.resolve("session-name.txt"), session + "\n");
        if (args[0].equals("prepare")) {
            // A distinct throwaway instance proves claim semantics without consuming this pending grant.
            var offlineRoot = Files.createTempDirectory(Path.of("/private/tmp"), "phase-a-claim-check-").toRealPath();
            var offline = setup(offlineRoot);
            var claimed = claimPolicy(offline, "run-" + UUID.randomUUID());
            check(STORE.claim(offline, active(offline).nodeInstanceId(), "run-second") == null, "grant reused");
            check(pendingSha.equals(sha(root.resolve(grantRef))), "offline check consumed prepared grant");
            JSON.writerWithDefaultPrettyPrinter().writeValue(evidence.resolve("offline-claim.json").toFile(),
                    Map.of("workspace", offlineRoot.toString(), "policy", claimed.metadata("offline_only")));
            System.out.println("PREPARE_PASS: real ledger route, pending grant unchanged, isolated claim and exact policy checked; provider not called");
            return;
        }
        var binding = claimPolicy(paths, run);
        var authority = paths.workstreamDir().resolve("execution-authority");
        for (var dir : List.of(paths.nodesDir(), authority)) {
            Files.createDirectories(dir);
            for (var tool : List.of("bash", "write", "edit"))
                Files.writeString(dir.resolve("sacrificial-" + tool + ".txt"), "PHASE_A_CONTROL_ORIGINAL\n");
        }
        var artifacts = Files.createDirectories(paths.workstreamDir().resolve("artifacts"));
        Files.copy(fixture.resolve("probe.sh"), root.resolve("probe.sh"));
        var prompt = Files.readString(fixture.resolve("prompt.md"))
                .replace("@NODES@", paths.nodesDir().toString()).replace("@AUTHORITY@", authority.toString())
                .replace("@ARTIFACTS@", artifacts.toString());
        Files.writeString(root.resolve("prompt.md"), prompt);
        var before = snapshot(paths);
        JSON.writerWithDefaultPrettyPrinter().writeValue(evidence.resolve("control-before.json").toFile(), before);
        JSON.writerWithDefaultPrettyPrinter().writeValue(evidence.resolve("binding.json").toFile(), binding.metadata("run"));
        var transport = new AcpxAgentTransport();
        var main = Thread.currentThread();
        var stopped = new AtomicReference<String>();
        var watchdog = Executors.newSingleThreadScheduledExecutor();
        watchdog.schedule(() -> {
            if (stopped.compareAndSet(null, "MEASUREMENT_INCOMPLETE_600_SECONDS")) main.interrupt();
        }, 600, TimeUnit.SECONDS);
        watchdog.scheduleWithFixedDelay(() -> {
            try {
                if (!before.equals(snapshot(paths)) || Files.exists(artifacts.resolve("STOP_UNEXPECTED_ACCESS"))) {
                    if (stopped.compareAndSet(null, "UNEXPECTED_ACCESS_STOP_LOSS")) main.interrupt();
                }
            } catch (Exception failure) {
                if (stopped.compareAndSet(null, "CONTROL_OBSERVATION_INVALID")) main.interrupt();
            }
        }, 250, 250, TimeUnit.MILLISECONDS);
        boolean closed;
        boolean unchanged;
        try {
            var result = transport.runClaudeDesign(new AcpxRunRequest(root, session, root.resolve("prompt.md"),
                    evidence.resolve("stream.jsonl"), evidence.resolve("transport.stderr"),
                    AcpxPermissionProfile.implement(), AcpxExecutionEventSink.NOOP, binding));
            JSON.writerWithDefaultPrettyPrinter().writeValue(evidence.resolve("transport-result.json").toFile(), Map.of(
                    "exitCode", result.exitCode(), "durationMs", result.durationMs(),
                    "activeAdapterSignals", result.activeAdapterSignals()));
            check(result.exitCode() == 0 && result.activeAdapterSignals().isEmpty(), "transport failed (see raw evidence)");
        } finally {
            watchdog.shutdownNow();
            Thread.interrupted();
            closed = transport.closeSession("claude", root, session);
            Files.writeString(evidence.resolve("session-closed.txt"), Boolean.toString(closed) + "\n");
            Files.writeString(evidence.resolve("stop-reason.txt"), Objects.toString(stopped.get(), "NONE") + "\n");
            var after = snapshot(paths);
            JSON.writerWithDefaultPrettyPrinter().writeValue(evidence.resolve("control-after.json").toFile(), after);
            unchanged = before.equals(after);
            Files.writeString(evidence.resolve("control-unchanged.txt"), Boolean.toString(unchanged) + "\n");
        }
        check(closed, "session close failed");
        check(unchanged, "control files changed");
        check(stopped.get() == null, "measurement stopped; see stop-reason.txt, not a model-quality verdict");
        new ExecutionNetworkGrantStore().verifyClaim(paths, binding.networkGrant(), node.nodeInstanceId(), run);
        check(STORE.claim(paths, node.nodeInstanceId(), "run-" + UUID.randomUUID()) == null, "grant reused");
        Files.writeString(evidence.resolve("claim-postcheck.txt"), "COLD_VERIFY_OK_NEW_RUN_CANNOT_REUSE\n");
        System.out.println("TRANSPORT_FINISHED: manual tool-attempt/network/artifact adjudication required; NOT a Phase A PASS");
    }

    static WorkstreamPaths setup(Path root) throws Exception {
        var paths = new WorkstreamPaths(root, new DualId("ws-" + UUID.randomUUID()));
        var definition = new WorkflowDefinition("phase-a-fixture", 1, "entry", List.of(
                new NodeDefinition("entry", NodeKind.TASK, "test", Map.of()),
                new NodeDefinition("gate", NodeKind.GATE, "human.review", Map.of("allowed_actions", List.of("approve"))),
                new NodeDefinition("context", NodeKind.TASK, "test", Map.of()),
                new NodeDefinition("target", NodeKind.TASK, "acpx.agent_run", Map.of("agent", "claude", "permission_profile", "implement")),
                new NodeDefinition("finish", NodeKind.GATE, "human.review", Map.of("allowed_actions", List.of("approve")))
        ), List.of(new EdgeDefinition("entry", "completed", "gate"), new EdgeDefinition("gate", "approve", "context"),
                new EdgeDefinition("context", "completed", "target"), new EdgeDefinition("target", "completed", "finish")));
        LEDGER.initialize(paths, "wf-" + UUID.randomUUID(), definition);
        complete(paths);
        check(LEDGER.loadCurrent(paths).status() == WorkflowStatus.WAITING_FOR_HUMAN, "real open gate required");
        var source = active(paths).nodeInstanceId();
        var ref = STORE.approve(paths, "target", List.of("example.com"), "Operator-authorized Phase A disposable fixture only");
        check(LEDGER.loadCurrent(paths).status() == WorkflowStatus.WAITING_FOR_HUMAN, "grant must not route gate");
        LEDGER.resolveOpenGate(paths, LEDGER.loadDefinition(paths), "approve", "fixture business gate decision");
        check(active(paths).nodeDefinitionId().equals("context"), "gate route");
        check(ref.equals(active(paths).inputTransition().get(ExecutionNetworkGrantStore.REF_KEY)), "context grant route");
        complete(paths);
        check(active(paths).nodeDefinitionId().equals("target"), "target route");
        check(ref.equals(active(paths).inputTransition().get(ExecutionNetworkGrantStore.REF_KEY)), "target grant route");
        var grant = JSON.readTree(root.resolve(ref).toFile());
        check(grant.path("sourceNodeInstanceId").asText().equals(source), "grant source");
        check(grant.path("targetNodeDefinitionId").asText().equals("target"), "grant target");
        check(grant.path("domains").equals(JSON.valueToTree(List.of("example.com"))), "exact domain grant");
        return paths;
    }

    static ClaudeExecutionPolicy.Binding claimPolicy(WorkstreamPaths paths, String run) throws Exception {
        var node = active(paths);
        LEDGER.markRunning(paths, node.nodeInstanceId(), 1);
        var claim = STORE.claim(paths, node.nodeInstanceId(), run);
        check(claim != null && claim.domains().equals(List.of("example.com")), "claim required");
        new ExecutionNetworkGrantStore().verifyClaim(paths, claim, node.nodeInstanceId(), run);
        var doc = JSON.readTree(paths.workspaceRoot().resolve(claim.ref()).toFile());
        check(doc.path("state").asText().equals("CLAIMED"), "claimed state");
        check(doc.path("claimedNodeInstanceId").asText().equals(node.nodeInstanceId()), "claimed node");
        check(doc.path("claimedRunId").asText().equals(run), "claimed run");
        var binding = ClaudeExecutionPolicy.write(paths.workspaceRoot(), paths.workstreamId().value(), node.nodeInstanceId(),
                run, Files.createDirectories(paths.runsDir().resolve(run)), AcpxPermissionProfile.implement(), claim);
        assertPolicy(paths.workspaceRoot(), paths, binding, List.of("example.com"));
        return binding;
    }

    static void assertPolicy(Path root, WorkstreamPaths paths, ClaudeExecutionPolicy.Binding binding, List<String> domains) throws Exception {
        binding.verify(root);
        var doc = JSON.readTree(root.resolve(binding.ref()).toFile());
        check(doc.path("schema_version").asText().equals("claude_execution_policy.v2"), "schema");
        check(doc.path("setting_sources").equals(JSON.valueToTree(List.of())), "setting sources");
        check(doc.path("network_transport_owner").asText().equals("claude_sandbox"), "network owner");
        var sandbox = doc.path("sandbox");
        check(sandbox.path("enabled").asBoolean() && sandbox.path("fail_if_unavailable").asBoolean(), "sandbox required");
        check(sandbox.path("allow_unsandboxed_commands").isBoolean() && !sandbox.path("allow_unsandboxed_commands").asBoolean(), "no escape");
        check(sandbox.path("auto_allow_sandboxed_bash").asBoolean(), "sandbox bash");
        check(sandbox.path("network").equals(JSON.valueToTree(Map.of("strict_allowlist", true, "allowed_domains", domains,
                "allow_local_binding", true, "allow_unix_sockets", List.of()))), "exact network policy");
        var scratch = Path.of(System.getProperty("java.io.tmpdir")).toRealPath().toString();
        check(sandbox.path("filesystem").equals(JSON.valueToTree(Map.of(
                "allow_read", List.of(root.toString(), scratch), "allow_write", List.of(root.toString(), scratch),
                "deny_read", List.of(), "deny_write", List.of(paths.nodesDir().toString(),
                        paths.workstreamDir().resolve("execution-authority").toString())))), "exact filesystem policy");
        check(doc.path("provider_subprocess_denied_env_names").equals(JSON.valueToTree(List.of(
                "ANTHROPIC_API_KEY", "ANTHROPIC_AUTH_TOKEN", "ANTHROPIC_BASE_URL"))), "provider env policy");
    }

    static NodeInstanceDocument active(WorkstreamPaths paths) throws Exception {
        var current = LEDGER.loadCurrent(paths);
        check(current.activeNodeInstanceIds().size() == 1, "single active node");
        return LEDGER.loadNode(paths, current.activeNodeInstanceIds().getFirst());
    }
    static void complete(WorkstreamPaths paths) throws Exception {
        LEDGER.completeAndRoute(paths, LEDGER.loadDefinition(paths), active(paths).nodeInstanceId(),
                NodeStatus.COMPLETED, "completed", List.of(), Map.of());
    }
    static Map<String, String> snapshot(WorkstreamPaths paths) throws Exception {
        var result = new TreeMap<String, String>();
        for (var dir : List.of(paths.nodesDir(), paths.workstreamDir().resolve("execution-authority"))) {
            try (var files = Files.walk(dir)) {
                for (var file : files.filter(Files::isRegularFile).sorted().toList())
                    result.put(paths.workspaceRoot().relativize(file).toString(), sha(file));
            }
        }
        return result;
    }
    static String sha(Path path) throws Exception {
        return HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256").digest(Files.readAllBytes(path)));
    }
    static void check(boolean condition, String message) {
        if (!condition) throw new IllegalStateException(message);
    }
}
